RECURSIVE CTE:
	Basic Structure:
		Q1,2:
		WITH RECURSIVE employee_tree AS (
		
		-- Anchor: start with CEO
		SELECT id,name,0 as level
		FROM employees
		WHERE name = 'CEO'

		UNION ALL

		-- Recursive part: find employees under them
		SELECT e.id,e.name,et.level + 1
		FROM employees e
		JOIN employee_tree et
			ON e.manager_id = et.id
		)

		SELECT *
		FROM employee_tree;
		
		
		Q3:
		with recursive employee_tree as (
			select id,name,0 as level
			from employees
			where name = 'CEO'
			
			union all
			
			select e.id,e.name,et.level+1
			from employees e
			join employee_tree et
			on e.manager_id = et.id
		)
		select * from employee_tree
		where level > 0;

WINDOW FUNCTIONS:
	Basic Structure:
	
	FUNCTION() OVER (
		PARTITION BY ...
		ORDER BY ...
	)
	
	ROW_NUMBER(),RANK(),DENSE_RANK()	
	
		Q1:
		select name,department,salary,
		row_number() over (
			partition by department
			order by salary desc
		) as rn
		from employees;
		
		
		Q2:
		select name,department,salary,
		rank() over (
			partition by department
			order by salary desc
		) as rn
		from employees;
		
		ranks look like : 1 2 2 4
		
		
		Q3:
		select name,department,salary,
		dense_rank() over (
			partition by department
			order by salary desc
		) as rn
		from employees;
		
		ranks look like : 1 2 2 3
		
		
		Q4:
		with highest_paid as (
			select name,department,salary,
			rank() over (
				partition by department
				order by salary desc
			) as rn
			from employees
		)
		
		select name,department,salary
		from highest_paid
		where rn = 1;

Question
Using a CTE, find employees whose salary is greater than 70,000.

Solution:
	with emp as (
		select * from employees
		where salary > 70000
	)
	select * from emp;
	
Q2:
Find the average salary of each department using a CTE.

Expected:

department | avg_salary
-----------+-----------
IT         | ...
HR         | ...
Sales      | ...

Solution:
	with avg_sal as (
		select department, avg(salary) from employees
		group by department
	)
	select * from avg_sal;
	
	
Q3:
Using the same employees table:
Find departments whose average salary is greater than 70,000.
Use a CTE.
Take your time and write the complete query.

Solution:
	with hadAvg_sal as (
		select departments,avg(salary)
		as avgsal
		from employees
		group by department
		having avg(salary) > 70000
	)
	select * from hadAvg_sal;
	

Q4:
Tables:
employees
id | name | department | salary

departments
department | location

Using a CTE, find employees whose salary is greater than their department's average salary.
Return:
name | department | salary

Solution:
	with cte as(
		select department,avg(salary) as avg_sal
		from employees
		group by department
	)
	select e.name,e.department,e.salary
	from employees e
	join department ds
	on ds.department = e.department
	where e.salary > cte.avgsal;
